<h2 class="wp-block-heading"><strong>Objectives</strong></h2>
<!-- /wp:heading -->

<!-- wp:paragraph -->
<p>We want a unique blog website that can:</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Show featured articles on Homepage</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Article page</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Show all category sections</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Allow readers to signup to a newsletter</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Showcase our Instagram feed</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Ability to Search through content</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Guest authors</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Social Sharing and liking of posts</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Article ratings</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph -->
<p><strong>Platform<br></strong>Please design for responsive web. So we require desktop screens as well as mobile screens.</p>
<!-- /wp:paragraph -->

<!-- wp:image {"align":"center"} -->
<figure class="wp-block-image aligncenter"><img src="https://cdn-images-1.medium.com/max/1600/1*CAErrL6NoE9BoUTD_eYNsw.png" alt="Design Element"/></figure>
<!-- /wp:image -->

<!-- wp:paragraph -->
<p><strong>Limitations</strong><br>Page performance (Loading time)<br>Limited Screen size on mobile</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p><strong>Target audience</strong><br>Young people between 25 to 35</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p><strong>Pain points</strong><br>Searching for inspiration about lifestyle, health, news, music, movies, and gadgets (Our category sections)<br>Get bored with uninspiring content <br>Hate having to return to websites to check for new articles</p>
