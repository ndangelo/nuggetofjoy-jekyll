<h3 class="wp-block-heading">Objectives</h3>
<!-- /wp:heading -->

<!-- wp:heading {"level":3} -->
<h3 class="wp-block-heading">Create a responsive eCommerce website that can:</h3>
<!-- /wp:heading -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>&nbsp;Tell the brand’s story alongside the products (VERY IMPORTANT)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>&nbsp;Showcase the latest products and bestsellers on the Homepage</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>&nbsp;Access to the cart throughout the site</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>&nbsp;Show all category sections</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>&nbsp;Allow readers to sign up for an account</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>&nbsp;Ability to Search through content</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>&nbsp;Immerse the user into the ethos of the brand</li>
<!-- /wp:list-item --></ul>
